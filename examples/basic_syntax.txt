rule("[helicopter] controls")
{
	event
	{
		Ongoing - Each Player;
		All;
		All;
	}

	conditions
	{
		Event Player.Is_Flying == True;
	}

	actions
	{
		Set Gravity(Event Player, 0);
		Event Player.flyspeed = 0;
		While(Event Player.Is_Flying);
			If(!Is Button Held(Event Player, Button(Primary Fire)));
				Event Player.flyspeed = 10;
			Else;
				Event Player.flyspeed = 1;
			End;
			If(Throttle Of(Event Player) != Vector(Empty Array, Empty Array, Empty Array));
				Apply Impulse(Event Player, Cross Product(Up, Normalize(World Vector Of(Throttle Of(Event Player), Event Player, Rotation))),
					0.001, To World, Cancel Contrary Motion);
				Apply Impulse(Event Player, Cross Product(Down, Normalize(World Vector Of(Throttle Of(Event Player), Event Player, Rotation))),
					0.001, To World, Cancel Contrary Motion);
				Apply Impulse(Event Player, Direction Towards(Eye Position(Event Player), Eye Position(Event Player) + Normalize(World Vector Of(
					Throttle Of(Event Player), Event Player, Rotation))), Event Player.flyspeed, To World, Cancel Contrary Motion);
				If(!Is Button Held(Event Player, Button(Crouch)));
					Apply Impulse(Event Player, Up, 0.001, To World, Cancel Contrary Motion);
				End;
				If(!Is Button Held(Event Player, Button(Jump)));
					Apply Impulse(Event Player, Down, 0.001, To World, Cancel Contrary Motion);
				End;
			Else;
				If(Speed Of(Event Player) > 0.010);
					Apply Impulse(Event Player, Left, 0.001, To World, Cancel Contrary Motion);
					Apply Impulse(Event Player, Right, 0.001, To World, Cancel Contrary Motion);
					Apply Impulse(Event Player, Forward, 0.001, To World, Cancel Contrary Motion);
					Apply Impulse(Event Player, Backward, 0.001, To World, Cancel Contrary Motion);
					If(!Is Button Held(Event Player, Button(Crouch)));
						Apply Impulse(Event Player, Up, 0.001, To World, Cancel Contrary Motion);
					End;
					If(!Is Button Held(Event Player, Button(Jump)));
						Apply Impulse(Event Player, Down, 0.001, To World, Cancel Contrary Motion);
					End;
				End;
			End;
			If(Is Button Held(Event Player, Button(Jump)));
				Apply Impulse(Event Player, Up, Event Player.flyspeed, To World, Cancel Contrary Motion);
			Else If(Is Button Held(Event Player, Button(Crouch)));
				Apply Impulse(Event Player, Down, Event Player.flyspeed, To World, Cancel Contrary Motion);
			End;
			Wait(0.016, Ignore Condition);
		End;
		Set Gravity(Event Player, 100);
	}
}

rule("[Sunstroke][sub] calculate Sun movement")
{
	event
	{
		Subroutine;
		CalculateSunMovement;
	}

	actions
	{
		Event Player.Sun_Q = Vector(X Component Of(Throttle Of(Event Player)) * -1, 0, Z Component Of(Throttle Of(Event Player)));
		If(Is Button Held(Event Player, Button(Jump)));
			Event Player.Sun_Q -= Vector(0, 1, 0);
		Else If(Is Button Held(Event Player, Button(Crouch)));
			Event Player.Sun_Q += Vector(0, 1, 0);
		End;
		If(Z Component Of(Throttle Of(Event Player)) == -1);
			Event Player.SunMovePosition = Direction From Angles(Angle Difference(Horizontal Angle From Direction(Event Player.Sun_Q),
				Horizontal Facing Angle Of(Event Player)), Angle Difference(Vertical Angle From Direction(Event Player.Sun_Q),
				Vertical Facing Angle Of(Event Player) * -1));
		Else If(Throttle Of(Event Player) == Left || Throttle Of(Event Player) == Right);
			Event Player.SunMovePosition = Direction From Angles(Angle Difference(Horizontal Angle From Direction(Event Player.Sun_Q),
				Horizontal Facing Angle Of(Event Player)), Angle Difference(Vertical Angle From Direction(Event Player.Sun_Q), 0));
		Else;
			Event Player.SunMovePosition = Direction From Angles(Angle Difference(Horizontal Angle From Direction(Event Player.Sun_Q),
				Horizontal Facing Angle Of(Event Player)), Angle Difference(Vertical Angle From Direction(Event Player.Sun_Q),
				Vertical Facing Angle Of(Event Player)));
		End;
	}
}

rule("[Sunstroke][sub] check Sun Collision")
{
	event
	{
		Subroutine;
		SunCollision;
	}

	actions
	{
		Event Player.Sun_R = Ray Cast Hit Position(Event Player.SunPosition, Event Player.SunPosition + Event Player.SunMovePosition * 3,
			All Players(All Teams), Event Player, True);
		If(Distance Between(Event Player.SunPosition, Event Player.Sun_R) < 3);
			For Player Variable(Event Player, Sun_Q, 20, 50, 15);
				"-Q degree vertical"
				Event Player.SunMovePosition = Direction From Angles(Horizontal Angle From Direction(Event Player.SunMovePosition),
					Vertical Angle From Direction(Event Player.SunMovePosition) - Event Player.Sun_Q);
				Skip If(Vertical Angle From Direction(Event Player.SunMovePosition) < -89.980 + Event Player.Sun_Q, 2);
				Event Player.Sun_R = Ray Cast Hit Position(Event Player.SunPosition, Event Player.SunPosition + Event Player.SunMovePosition * 3,
					All Players(All Teams), Event Player, True);
				Abort If(Distance Between(Event Player.SunPosition, Event Player.Sun_R) >= 3);
				"+Q degree vertical"
				Event Player.SunMovePosition = Direction From Angles(Horizontal Angle From Direction(Event Player.SunMovePosition),
					Vertical Angle From Direction(Event Player.SunMovePosition) + Event Player.Sun_Q);
				Skip If(Vertical Angle From Direction(Event Player.SunMovePosition) > 89.980 - Event Player.Sun_Q, 2);
				Event Player.Sun_R = Ray Cast Hit Position(Event Player.SunPosition, Event Player.SunPosition + Event Player.SunMovePosition * 3,
					All Players(All Teams), Event Player, True);
				Abort If(Distance Between(Event Player.SunPosition, Event Player.Sun_R) >= 3);
				"-Q degree horizontal"
				Event Player.SunMovePosition = Direction From Angles(Horizontal Angle From Direction(Event Player.SunMovePosition)
					+ Event Player.Sun_Q, Vertical Angle From Direction(Event Player.SunMovePosition) - Event Player.Sun_Q);

					+ Event Player.Sun_Q, Vertical Angle From Direction(Event Player.SunMovePosition));
			End;
			Stop Chasing Player Variable(Event Player, SunPosition);
			Event Player.SunIsMoving = False;
			Wait(0.032, Ignore Condition);

					Destination and Rate);

	}
