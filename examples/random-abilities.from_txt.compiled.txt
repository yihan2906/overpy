rule ("track interact press count") {
    event {
        Ongoing - Each Player;
        All;
        All;
    }
    conditions {
        Is Button Held(Event Player, Button(Interact)) == True;
        (Event Player).interact_is_held == False;
    }
    actions {
        Set Player Variable(Event Player, interact_is_held, True);
        Modify Player Variable(Event Player, interact_press_count, Add, 1);
    }
}

rule ("track interact release") {
    event {
        Ongoing - Each Player;
        All;
        All;
    }
    conditions {
        Is Button Held(Event Player, Button(Interact)) == False;
        (Event Player).interact_is_held == True;
    }
    actions {
        Set Player Variable(Event Player, interact_is_held, False);
    }
}

rule ("end death match") {
    event {

        Is Assembling Heroes == True;
        Array Contains(Array(Game Mode(Deathmatch), Game Mode(Team Deathmatch)), Current Game Mode) == True;
    }
    actions {
        Set Match Time(0);
    }
}

        All;

    }
    conditions {
        (Event Player).Is_Flying == True;
    }
    actions {
        Set Gravity(Event Player, 0);
        Set Player Variable(Event Player, flyspeed, 0);
        While((Event Player).Is_Flying);
            If(Not(Is Button Held(Event Player, Button(Primary Fire))));
                Set Player Variable(Event Player, flyspeed, 10);
            Else;
                Set Player Variable(Event Player, flyspeed, 1);
            End;
            If(Compare(Throttle Of(Event Player), !=, Vector(Empty Array, Empty Array, Empty Array)));
                Apply Impulse(Event Player, Cross Product(Up, Normalize(World Vector Of(Throttle Of(Event Player), Event Player, Rotation))), 0.001, To World, Cancel Contrary Motion);
                Apply Impulse(Event Player, Cross Product(Down, Normalize(World Vector Of(Throttle Of(Event Player), Event Player, Rotation))), 0.001, To World, Cancel Contrary Motion);
                Apply Impulse(Event Player, Direction Towards(Eye Position(Event Player), Add(Eye Position(Event Player), Normalize(World Vector Of(Throttle Of(Event Player), Event Player, Rotation)))), (Event Player).flyspeed, To World, Cancel Contrary Motion);
            End;
            If(And(Compare(Throttle Of(Event Player), !=, Vector(Empty Array, Empty Array, Empty Array)), Not(Is Button Held(Event Player, Button(Crouch)))));
                Apply Impulse(Event Player, Up, 0.001, To World, Cancel Contrary Motion);
            End;
            If(And(Compare(Throttle Of(Event Player), !=, Vector(Empty Array, Empty Array, Empty Array)), Not(Is Button Held(Event Player, Button(Jump)))));
                Apply Impulse(Event Player, Down, 0.001, To World, Cancel Contrary Motion);
            End;
            If(And(Compare(Throttle Of(Event Player), ==, Vector(Empty Array, Empty Array, Empty Array)), Compare(Speed Of(Event Player), >, 0.01)));
                Apply Impulse(Event Player, Left, 0.001, To World, Cancel Contrary Motion);
                Apply Impulse(Event Player, Right, 0.001, To World, Cancel Contrary Motion);
                Apply Impulse(Event Player, Forward, 0.001, To World, Cancel Contrary Motion);
                Apply Impulse(Event Player, Backward, 0.001, To World, Cancel Contrary Motion);
            End;
            If(And(And(Compare(Throttle Of(Event Player), ==, Vector(Empty Array, Empty Array, Empty Array)), Compare(Speed Of(Event Player), >, 0.01)), Not(Is Button Held(Event Player, Button(Crouch)))));
                Apply Impulse(Event Player, Up, 0.001, To World, Cancel Contrary Motion);
            End;
            If(And(And(Compare(Throttle Of(Event Player), ==, Vector(Empty Array, Empty Array, Empty Array)), Compare(Speed Of(Event Player), >, 0.01)), Not(Is Button Held(Event Player, Button(Jump)))));
                Apply Impulse(Event Player, Down, 0.001, To World, Cancel Contrary Motion);
            End;
            If(Is Button Held(Event Player, Button(Jump)));
                Apply Impulse(Event Player, Up, (Event Player).flyspeed, To World, Cancel Contrary Motion);
            Else If(Is Button Held(Event Player, Button(Crouch)));
                Apply Impulse(Event Player, Down, (Event Player).flyspeed, To World, Cancel Contrary Motion);
            End;
            Wait(0.016, Ignore Condition);
        End;
        Set Gravity(Event Player, 100);
    }

            Set Player Variable(Event Player, SunMovePosition, Direction From Angles(Angle Difference(Horizontal Angle From Direction((Event Player).Sun_Q), Horizontal Facing Angle Of(Event Player)), Angle Difference(Vertical Angle From Direction((Event Player).Sun_Q), Vertical Facing Angle Of(Event Player))));
        End;
    }
